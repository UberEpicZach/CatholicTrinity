# "The Saxon Revolt" # Event for Otto of Bavaria and Heinrich IV, maybe everyone in the struggle region?

tct_neutral_struggle.0009 = {
	type = character_event
	title = tct_neutral_struggle.0009.t
	desc = tct_neutral_struggle.0009.desc

	theme = war
	override_background = {
		reference = ce1_legendary_battlefield
	}
	left_portrait = {
		character = scope:otto
		animation = marshal_shield
		camera = camera_event_scheme_vs_center_positioned_left
	}
	right_portrait = {
		character = scope:magnus
		animation = marshal_shield
		camera = camera_event_scheme_vs_center_look_left
	}

	immediate = {
		character:1418 = {
			save_scope_as = magnus
		}
		scope:magnus = {
			add_character_flag = {
				days = 130
				flag = wear_armor
			}
			culture = {
				save_scope_as = culture
			}
		}
		scope:otto = {
			add_character_flag = {
				days = 130
				flag = wear_armor
			}
		}

		if = {
			limit = {
				NOT = {
					exists = global_var:saxon_revolt_fired
				}
			}
			set_global_variable = saxon_revolt_fired

			scope:emperor = {
				if = {
					limit = {
						NOT = {
							OR = {
								has_realm_law = crown_authority_1
								has_realm_law = crown_authority_2
								has_realm_law = crown_authority_3
							}
						}
					}
					add_realm_law = crown_authority_1
				}
			}

			scope:magnus = { 
				create_faction = {
					type = liberty_faction
					target = title:e_hre.holder
				}
				joined_faction = {
					save_scope_as = faction
				}
			}
			scope:emperor = {
				every_vassal_or_below = {
					limit = { 
						culture = scope:magnus.culture 
						NOT = { this = scope:magnus }
						exists = scope:magnus.joined_faction
						can_join_faction = scope:magnus.joined_faction
					}
					custom = every_culture_vassal
					random = {
						chance = 40
						if = {
							limit = {
								exists = scope:magnus.joined_faction
							}
							join_faction = scope:magnus.joined_faction
						}
					}
				}
			}
			scope:magnus.joined_faction = {
				faction_start_war = { title = title:e_hre }
				set_casus_belli = first_saxon_revolt_faction_war
			}
		}

		struggle:investiture_controversy_struggle = {
			every_involved_ruler = {
				limit = {
					NOT = {
						this = scope:otto 
						this = scope:magnus
						this = title:e_hre.holder
						has_character_flag = had_tct_neutral_struggle_0009
					}
				}
				trigger_event = { days = 1 id = tct_neutral_struggle.0009 }
				add_character_flag = had_tct_neutral_struggle_0009
			}
		}
	}

	option = {
		name = tct_neutral_struggle.0009.a
		trigger = {
			this = scope:otto
		}

		scope:magnus = {
			random_character_war = {
				limit = {
					using_cb = liberty_faction_war
					primary_defender = scope:emperor
				}
				save_scope_as = faction_war
			}
			scope:faction_war = {
				set_casus_belli = first_saxon_revolt_faction_war
			}
		}

		# This needs to occur after the war begins, or the troops can't spawn correctly.
		faction_spawn_member_county_armies_effect = {
			FACTION = scope:faction
			ARMY_OWNER = scope:magnus
			PEASANT_ARMY_NAME = populist_faction_event_troops
		}
