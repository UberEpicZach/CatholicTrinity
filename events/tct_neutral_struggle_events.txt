				}

				}

					set_casus_belli = first_saxon_revolt_faction_war
				}
				if = {
					limit = {
						exists = scope:faction_war
					}

						levies = 4000
						location = scope:magnus.location
						war = scope:faction_war
						name = populist_faction_event_troops
					}
				}
			}

			}
		}

					}
				}

			trigger = {
				this = scope:nephew.mother
			}

		}

			trigger = {
				this = character:7757
			}
			animation = anger
		}

			trigger = {
				always = yes
			}
			animation = sadness
		}
	}

		add_character_flag = had_tct_neutral_struggle_2001
		if = {
			limit = { is_married = yes }
			primary_spouse = { save_scope_as = female_party }
		}

			primary_heir = { save_scope_as = female_party }
		}
		primary_title = {
			ordered_title_heir = { # Pick first male heir
				limit = {
					is_male = yes
				}
				position = 1
				save_scope_as = first_male_heir

			if = { # If first is someone else's heir, pick the next one

						OR = {
							any_pretender_title = {
								exists = holder
								NOT = { holder = root }
							}
							any_heir_title = {
								exists = holder
								NOT = { holder = root }
							}
						}
					}

					save_scope_as = nephew
				}
				clear_saved_scope = first_male_heir
			}
		}
	}

					opinion = -50
				}
			}
			show_as_tooltip = {
				random_list = {
					100 = {
						show_chance = no
						desc = hunchback_dread_if_female_party_rejects_tt
						if = {
							limit = {
								OR = {
									scope:female_party = character:7757
									matrilinear_marriage = yes
								}
							}
							show_as_tooltip = {
								tct_neutral_struggle_2001_title_claims_effect = { SCOPED = scope:female_party }
							}
						}
					}
					100 = {
						show_chance = no
						desc = hunchback_dread_if_female_party_accepts_tt
					}
				}
			}
		}
		stress_impact = {
			ambitious = medium_stress_impact_gain
			paranoid = medium_stress_impact_gain
		}
	}

		name = tct_neutral_struggle.2001.b
		trigger = {
			NOT = { has_trait = paranoid }
		}
		scope:female_party = {
			add_opinion = {
