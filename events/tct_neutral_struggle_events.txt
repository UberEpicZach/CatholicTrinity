# Offer from Emperor, for Welf to Divorce Wife / Split from Nordheim's, In exchange, get the Duchy of Bavaria

tct_neutral_struggle.1003 = {
	type = character_event
	title = tct_neutral_struggle.1003.t
	desc = tct_neutral_struggle.1003.desc

	theme = realm
	override_background = {
		reference = throne_room
	}
	left_portrait = {
		character = root
		animation = thinking
	}
	lower_left_portrait = {
		character = scope:spouse
	}
	lower_center_portrait = {
		character = scope:otto
	}
	right_portrait = {
		character = scope:imperial_councillor
		animation = writing
	}

	immediate = {
		random_spouse = {
			save_scope_as = spouse
		}
		scope:emperor.cp:councillor_chancellor = {
			save_scope_as = imperial_councillor
		}
	}

	option = {
		name = tct_neutral_struggle.1003.a
		
		divorce_effect = {
			DIVORCER = root
			DIVORCEE = scope:spouse
		}

		if = {
			limit = {
				NOT = {
					scope:otto = {
						is_at_war_with = title:e_hre.holder
					}
				}
			}
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = yes
			}
	
			scope:otto = {
				every_held_title = {
					change_title_holder = {
						holder = root
						change = scope:change
					}
				}
				every_vassal = {
					change_liege = {
						liege = root
						change = scope:change
					}
				}
			}
			resolve_title_and_vassal_change = scope:change
	
			scope:otto = { 
				add_character_flag = force_ai_created_laamp 
				#depose_effect = { DEPOSER = scope:emperor } 
				create_landless_adventurer_title_effect = { REASON = flag:deposition FLAVOR_CHAR = title:e_hre.holder }
				hidden_effect = {
					add_pressed_claim = title:k_bavaria
					add_realm_law_skip_effects = camp_purpose_legitimists
				}
			}
	
			add_opinion = {
				target = title:e_hre.holder
				modifier = received_title_duchy
				opinion = 80
			}
	
			scope:emperor = {
				add_opinion = {
					target = root
					modifier = loyal_servant
					opinion = 50
				}
			}
		}
		else = {
			if = {
				limit = {
					scope:otto = {
						is_at_war_with = title:e_hre.holder
					}
				}
				scope:emperor = {
					every_character_war = {
						limit = {
							OR = {
								primary_attacker = scope:otto
								primary_defender = scope:emperor
							}
						}
						add_defender = scope:welf
					}
				}
				add_character_flag = promised_bavaria
			}
		}


		ai_chance = {
			base = 100
		}
	}

	option = {
		name = tct_neutral_struggle.1003.b

		hidden_effect = {
			character:7054 = { # For fun, if Welf doesn't take the deal, Frederich Hohenstaufen will
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = no
				}
		
				scope:otto = {
					every_held_title = {
						change_title_holder = {
							holder = character:7054 
							change = scope:change
						}
					}
					every_vassal = {
						change_liege = {
							liege = character:7054 
							change = scope:change
						}
					}
				}
				resolve_title_and_vassal_change = scope:change
	
				scope:otto = { 
					add_character_flag = force_ai_created_laamp 
					#depose_effect = { DEPOSER = scope:emperor } 
					create_landless_adventurer_title_effect = { REASON = flag:deposition FLAVOR_CHAR = title:e_hre.holder }
				}
	
				add_opinion = {

	title = tct_neutral_struggle.1051.t
	type = character_event
	desc = tct_neutral_struggle.1051.desc
	theme = war
	override_background = {
		reference = wilderness_scope
	}

	left_portrait = {
		character = root
		animation = fear
	}

	trigger = {
		has_character_flag = enable_saxon_peasants
		scope:old_location.county ?= {
			NOT = { culture = culture:saxon }
		}
		scope:new_location.county ?= {
			culture = culture:saxon
		}
		NOT = { has_character_flag = on_goining_ambush }
	}

	#Continue
	immediate = {
		add_character_flag = {
			days = 100 flag = on_goining_ambush
		}
		hidden_effect = {
			scope:army = { 
				location = { 
					save_scope_as = target_location 
					save_scope_as = background_wilderness_scope
				} 
			}
			#location = { save_scope_as = target_location }
			create_saxon_ambush = yes
		}
	}
	option = {
		name = tct_neutral_struggle.1051.a
		stress_impact = {
			craven = major_stress_impact_gain
		}
		spawn_saxon_army = { TARGET = root }
	}

	after = {
		scope:saxon_char = {
			assign_commander_trait_effect = yes
			ordered_army = {
				order_by = army_size
				assign_commander = scope:saxon_char
			}
		}
	}
}

		limit = {
			is_landless_type_title = no
			is_noble_family_title = no

				limit = {
					scope:actor = { has_claim_on = scope:target }
				}

			}
			tier >= tier_county
		}
		save_temporary_scope_as = title_temp
		$SCOPED$ = {
			if = {
				limit = {
					OR = {
						this = character:7757
						has_claim_on = scope:title_temp
					}
				}
				add_pressed_claim = scope:title_temp
			}
			else = {
				add_unpressed_claim = scope:title_temp
			}
		}
	}
}

# "A Hunchback's Dread" # Event for Godfrey the Hunchback
tct_neutral_struggle.2001 = {
	type = character_event
	title = tct_neutral_struggle.2001.t
	desc = {
		triggered_desc = {
			trigger = { always = yes }
			desc = tct_neutral_struggle.2001.desc
		}
		triggered_desc = {
			trigger = {
				is_married = yes
				scope:female_party = character:7757
			}
			desc = tct_neutral_struggle.2001.desc.matilda
		}
		triggered_desc = {
			trigger = {
				is_married = yes
				NOT = { scope:female_party = character:7757 }
			}
			desc = tct_neutral_struggle.2001.desc.wife
		}
		triggered_desc = {
			trigger = {
				scope:female_party = scope:nephew.mother
			}
			desc = tct_neutral_struggle.2001.desc.mother
		}
	}
	
	trigger = {
		NOT = { has_character_flag = had_tct_neutral_struggle_2001 }
		primary_heir = {
			is_female = yes
			NOT = { is_child_of = root }
		}
		primary_title = {
			any_title_heir = {
				is_male = yes
			}
		}
	}

	theme = realm
	override_background = {
		reference = study
	}
	left_portrait = {
		character = root
		animation = paranoia
	}
	right_portrait = {
		character = scope:female_party
		triggered_animation = {
			trigger = {
				this = scope:nephew.mother
			}
			animation = personality_content
		}
		triggered_animation = {
			trigger = {
				this = character:7757
			}
			animation = anger
		}
		triggered_animation = {
			trigger = {
				always = yes
			}
			animation = sadness
		}
	}
	lower_right_portrait = scope:nephew

	immediate = {
		add_character_flag = had_tct_neutral_struggle_2001
		if = {
			limit = { is_married = yes }
			primary_spouse = { save_scope_as = female_party }
		}
		else = {
			primary_heir = { save_scope_as = female_party }
		}
		primary_title = {
			ordered_title_heir = { # Pick first male heir
				limit = {
					is_male = yes
				}
				position = 1
				save_scope_as = first_male_heir
			}
			if = { # If first is someone else's heir, pick the next one
				limit = {
					scope:first_male_heir = {
						OR = {
							any_pretender_title = {
								exists = holder
								NOT = { holder = root }
							}
							any_heir_title = {
								exists = holder
								NOT = { holder = root }
							}
						}
					}
				}
				ordered_title_heir = {
					limit = {
						NOT = { this = scope:first_male_heir }
						is_male = yes
					}
					position = 2
					save_scope_as = nephew
				}
				clear_saved_scope = first_male_heir
			}
			else = { # First one is fine
				scope:first_male_heir = {
					save_scope_as = nephew
				}
				clear_saved_scope = first_male_heir
			}
		}
	}

	option = {
		name = tct_neutral_struggle.2001.a
		set_designated_heir_unsafe = scope:nephew
		tct_neutral_struggle_2001_title_claims_effect = { SCOPED = scope:nephew }
		if = {
			limit = {
				NOT = { scope:female_party = scope:nephew.mother }
			}
			scope:female_party ?= {
				if = {
					limit = { is_ruler = yes }
					trigger_event = tct_neutral_struggle.2002
				}
				add_opinion = {
					target = root
					modifier = insulted_opinion
					opinion = -50
				}
			}
			show_as_tooltip = {
				random_list = {
					100 = {
						show_chance = no
						desc = hunchback_dread_if_female_party_rejects_tt
						if = {
							limit = {
								OR = {
									scope:female_party = character:7757
									matrilinear_marriage = yes
								}
							}
							show_as_tooltip = {
								tct_neutral_struggle_2001_title_claims_effect = { SCOPED = scope:female_party }
							}
						}
					}
					100 = {
						show_chance = no
						desc = hunchback_dread_if_female_party_accepts_tt
					}
				}
			}
		}
		stress_impact = {
			ambitious = medium_stress_impact_gain
			paranoid = medium_stress_impact_gain
		}
	}

	option = {
		name = tct_neutral_struggle.2001.b
		trigger = {
			NOT = { has_trait = paranoid }
		}
		scope:female_party = {
			add_opinion = {
				target = root
				modifier = grateful_opinion
				opinion = 25
			}
		}
		stress_impact = {
			just = medium_stress_impact_gain
			ambitious = medium_stress_impact_gain
		}

		ai_chance = {
			base = 0
		}
	}

	after = {
		# Bypass Know Thyself in case it is picked after this event
		add_character_flag = {
			flag = know_thyself_perk_delay_period
			days = 390
		}
		add_character_flag = know_thyself_has_triggered_delay
		if = {
			limit = {
				has_perk = know_thyself_perk
				NOT = { has_character_flag = had_death_management_9998_recently }
			}
			trigger_event = {
				id = death_management.9998
				days = { 15 45 }
			}
		}
		else = {
			trigger_event = {
				id = tct_neutral_struggle.2003
				days = { 340 380 } # Similar to death_management.9999
			}
		}
	}
}
