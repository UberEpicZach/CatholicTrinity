namespace = TCT_saint_days

TCT_saint_days.0001 = {
	type = character_event
	title = TCT_saint_days.0001.t
	desc = {
		random_valid = {
			desc = TCT_saint_days.0001.desc.a
			desc = TCT_saint_days.0001.desc.b
			desc = TCT_saint_days.0001.desc.c
			desc = TCT_saint_days.0001.desc.d
		}
	}
	theme = family
	
	left_portrait = {
		character = scope:family
		animation = happiness
		trigger = {
			exists = scope:family
		}
	}
	
	right_portrait = {
		character = root
		animation = ecstasy
	}

	lower_left_portrait = {
		character = scope:localsaint
	}

	trigger = {
		saintdays_event1trigger = yes
	}
	
	immediate = {
		if = { # Get random family member for portrait
			limit = {
				any_close_family_member = {
					count >= 1
					is_alive = yes
					is_in_the_same_court_as = root
					opinion = {
						target = root
						value >= 0
					}
				}
			}
			random_close_family_member = {
				limit = {
					is_in_the_same_court_as = root
					opinion = {
						target = root
						value >= 0
					}
				}
				weight = {
					base = 1
					modifier = {
						add = 10
						is_child_of = root
					}
				}
				save_scope_as = family
			}
		}
		if = { # Get husband as stand-in for portrait
			limit = {
				exists = primary_spouse
				is_in_the_same_court_as = root
			}
			if = {
				limit = {
					NOT = { exists = scope:family }
				}
				primary_spouse = { save_scope_as = family }
			}
			else = {
				random_List = {
					50 = {
						primary_spouse = {
							save_scope_as = family
						}
					}
					50 = {}
				}
			}
		}
		random_sub_realm_county = {
			limit = {
				saintdays_province = yes
			}
			save_scope_as = saint_day_province
			
			set_variable = {
				name = saint_day_cooldown
				years = 10
			}
		}
		scope:saint_day_province.county.var:saint_day_saint = {
			save_scope_as = localsaint
		}
	}

	option = { #ok
		name = TCT_saint_days.0001.b
	}

	option = { #Bah, humbug!
		name = TCT_saint_days.0001.d
		add_character_flag = TCT_humbug
		custom_tooltip = TCT_saint_days.0001.d.tooltip
	}
}
