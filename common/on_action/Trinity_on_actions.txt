# On Game Start On Actions
tct_on_start = {
	effect = {
		# Map Mode Setup
		set_global_variable = {
			name = custom_map_mode
			value = flag:none
		}

		### Pentarchy setups
		# title:k_orthodox.holder = {
		# 	set_variable = tct_ecumenical_patriarch
		# }
		# faith:orthodox = {
		# 	add_doctrine = doctrine_autocephaly_head
		# }
		every_living_character = {
			limit = {
				OR = {
					has_title = title:k_papal_state
					has_title = title:d_rome_patriarchate
					has_title = title:k_orthodox
					has_title = title:d_coptic_papacy
					has_title = title:d_antioch_patriarchate
					has_title = title:d_jerusalem_patriarchate
				}
			}
			change_government = theocracy_government
		}

		# Catholicism Setup
		faith:catholic = {
			add_doctrine = doctrine_independent_papacy_head
			add_doctrine = doctrine_clerical_succession_investiture_law_appointment
			if = {
				limit = { has_game_rule = tct_cosanguinity_on }
				if = {
					limit = { has_doctrine = doctrine_consanguinity_aunt_nephew_and_uncle_niece }
					remove_doctrine = doctrine_consanguinity_aunt_nephew_and_uncle_niece
				}
				if = {
					limit = { NOT = { has_doctrine = doctrine_consanguinity_restricted } }
					add_doctrine = doctrine_consanguinity_restricted
				}
			}
		}

		if = {
			limit = { has_game_rule = latin_learning_default }
			title:k_papal_state = {
				if = {
					limit = {
						exists = holder
						has_dlc_feature = royal_court
					}
					holder = {
						set_court_language = language_latin
						if = {
							limit = { NOT = { knows_court_language_of = this } }
							learn_court_language_of = this
						}
					}
				}
			}
		}
		if = {
			limit = {
				OR = {
					has_game_rule = latin_learning_default
					has_game_rule = latin_learning_priests
				}
			}
			faith:catholic = { 
				every_faith_character = { 
					limit = { 
						is_clergy = yes 
						NOT = { 
							knows_language = language_latin 
						} 
					} 
					learn_language = language_latin 
				}
			}
		}
		set_global_variable = {
			name = Trinity_is_loaded
			value = yes
		}
		title:k_papal_state.holder = {
			set_variable = was_once_cardinal
			add_trait = pope_trait #ZGC
		}

		# Title Setup
		if = {
			limit = {
				exists = title:d_captain_general_church.holder
			}
			title:d_captain_general_church.holder = {
				get_title = title:c_tivoli

				create_title_and_vassal_change = {
					type = granted #or whatever
					save_scope_as = change
				}
				
				change_liege = {
					liege = title:k_papal_state.holder
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
			}
		}

		title:b_paterno.title_province = { set_holding_type = castle_holding }

		# Struggle Setup
		if = {
			limit = {
				exists = struggle:investiture_controversy_struggle
			}
			title:e_hre = { set_variable = struggle_block_dissolution_faction }
		}
	}
}

on_cardinal_preinitialize = {
	# Set back-end configuration data 
	effect = {
		if = { # Vanilla parameters
			limit = {
				exists = faith:catholic
			}
			set_global_variable = {
				name = gui_storage
				value = religion:christianity_religion
			}
			set_global_variable = {
				name = the_catholic
				value = faith:catholic
			}
			set_global_variable = {
				name = insular_celts
				value = faith:insular_celtic
			}
			set_global_variable = {
				name = k_papal_state
				value = title:k_papal_state
			}
			set_global_variable = {
				name = ideal_cardinal_location
				value = title:c_roma
			}
			every_ruler = { 
				limit = { 
					has_faith = faith:catholic 
					NOT = { 
						exists = var:cardinal_player_candidate 
						has_government = mercenary_government 
						has_government = theocracy_government 
					}
					primary_title.title_held_years > 1
					is_at_war = no
					is_ruler = yes
					is_landed = yes
					is_cardinal = no
					is_pope = no
					is_alive = yes
					exists = this
				} 
				select_highest_scoring_character_cardinal_candidate = yes 
			}
		}
		global_var:the_catholic = {
			set_variable = uses_cardinal_module
		}
		global_var:insular_celts = {
			set_variable = uses_cardinal_module
		}
	}
}

					save_scope_as = ghw_sponsor
				}

					set_war_declarer = scope:ghw_sponsor

						limit = {
							NOT = { has_pledged_attacker = scope:ghw_sponsor }
						}
						pledge_attacker = scope:ghw_sponsor
					}
				}
			}
		}
	}

on_theocrat_investiture_death = {
	trigger = {
		exists = primary_title
		primary_title = {
			has_variable = theocratic_successor
		}
	}
	effect = {
		liege = { save_scope_as = important_liege }
		primary_title = { var:theocratic_successor = { save_scope_as = theocratic_inheritor } remove_variable = theocratic_successor }
		TCT_theocrat_investiture_opinion = {
			SUCCESSOR = scope:theocratic_inheritor
		}
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = title_change
			add_claim_on_loss = no
		}
		every_held_title = {
			change_title_holder = {
				holder = scope:theocratic_inheritor
				change = scope:title_change
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		scope:theocratic_inheritor = {
			remove_variable = theocratic_successor
			change_government = theocracy_government
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = change
			}
			change_liege = {
				liege = scope:important_liege
				change = scope:change
			}
			resolve_title_and_vassal_change = scope:change
		}
		scope:important_liege = {
			faith.religious_head = {
				add_opinion = {
					target = scope:important_liege
					opinion = -20
					modifier = TCT_appointed_bishop_investiture_bad
				}
			}
		}
	}
}

on_theocratic_investiture_death_no_successor = {
	trigger = {
		exists = primary_title
		primary_title = {
			NOT = {
				has_variable = theocratic_successor
			}
		}
		government_has_flag = government_is_theocracy
		top_liege = {
			has_realm_law = free_investiture
		}
	}
	effect = {
		if = {
			limit = {
				is_ai = yes
			}
			if = {
				limit = {
					any_character_struggle	 = {
						involvement = involved
						phase_has_catalyst = tct_appointed_bishop_investiture
					}
				}
				every_character_struggle = {
					involvement = involved
					activate_struggle_catalyst = {
						catalyst = tct_appointed_bishop_investiture
						character = root
					}
				}
			}
			faith.religious_head = {
				add_opinion = {
					target = liege
					opinion = -20
					modifier = TCT_appointed_bishop_investiture_bad
				}
			}
			add_opinion = {
				target = liege
				opinion = 40
				modifier = TCT_appointed_bishop_investiture_good
			}
		}
		else = {
			save_scope_as = dying_bishop
			liege = {
				trigger_event = TCT_religious_events.15
			}
		}
	}
}

on_gain_papacy = {
	trigger = {
		scope:title = title:k_papal_state
	}
	effect = {
		add_trait = pope_trait
		if = { limit = { is_lowborn = yes } title:k_papal_state.holder = { set_house = house:house_SanctiPetriFiliis } }
		if = {
			limit = { 
				has_trait = cardinal
			}
			change_trait_rank = {
				trait = cardinal
				rank = -3
			}
		}
		if = { limit = { has_trait = TCT_antipope } TCT_remove_antipope = yes }
		if = {
			limit = {
				any_held_title = {
					has_variable = antipope_title
				}
			}
			every_held_title = {
				limit = { has_variable = antipope_title }
				root = { destroy_title = prev }
			}
		}
	}
}

Trinity_on_war_started = {
	effect = {
		if = {
			limit = {
				exists = title:k_papal_state.holder
				scope:attacker = {
					has_faith = faith:catholic
					NOT = { has_government = theocracy_government }
					NOT = { has_trait = excommunicated }
				}
				
				scope:defender = {
					has_faith = faith:catholic
					has_government = theocracy_government
					NOT = { this = title:k_papal_state.holder }
				}
			}
			scope:attacker = {
				save_scope_as = BishopAttacker
			}
			scope:defender = {
				save_scope_as = BishopyDefender
			}
			title:k_papal_state.holder = {
				trigger_event = {
					id = TCT_religious_events.56
					days = 3
				}
			}
		}
	}
}
