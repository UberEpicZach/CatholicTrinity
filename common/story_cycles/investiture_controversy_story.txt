investiture_controversy_story = {
	on_setup = { # HRE Emperor Should be the story owner
		story_owner = { # Add Character Modifiers
			
		}
	}

	on_end = {
		debug_log = "investiture_controversy_story story ended on:"
		debug_log_date = yes
	}

	on_owner_death = {	
		story_owner = {
			if = {
				limit = {
					AND = {
						exists = player_heir
						exists = title:e_hre.holder
					}
				}
				player_heir = {
					limit = {
						is_primary_heir_of = title:e_hre	
					}
					save_scope_as = new_story_owner # pass chain to appropriate target
				}
			}
			else = {
				limit = {
					AND = {
						exists = title:e_hre.holder
					}
				}
				title:e_hre.holder = {
					save_scope_as = new_story_owner # pass chain to appropriate target
				}		
			}
		}
		if = {
			limit = { exists = scope:new_story_owner }
			make_story_owner = scope:new_story_owner

			debug_log = "Successfully replaced the investiture_controversy_story owner"
			debug_log_date = yes
			debug_log_scopes = yes
		}
	}

	effect_group = { # Story handling in case of succession weirdness
		days = 180

		trigger = {
			story_owner = {
				NOT = {
					any_held_title = {
						this = title:e_hre
					}
				}
			}
		}

		triggered_effect = {
			trigger = { exists = title:e_hre }
			effect = {
				title:e_hre.holder = {
					make_story_owner = title:e_hre.holder
				}
			}
		}

		triggered_effect = {
			trigger = { 
				NOT = { 
					exists = title:e_hre 
				} 
			}
			effect = {
				end_story = yes
			}
		}
	}


	# Emperor's Appointment of an Anti-Pope
	effect_group = { # 
		days = 180

		trigger = {
			title:e_hre.holder = {
				OR = {
					any_vassal_or_below = {
						tct_possible_antipope_vassal_trigger = yes
						save_temporary_scope_as = antipope_vassal_temp
					}
					AND = {
						exists = character:73530
						character:73530 = { save_temporary_scope_as = antipope_vassal_temp }
					}
				}
				is_character_interaction_potentially_accepted = {
					recipient = scope:antipope_vassal_temp
					interaction = TCT_make_antipope_interaction
				}
				any_character_struggle = {
					is_struggle_type = investiture_controversy_struggle
				}
			}
		}

		triggered_effect = {
			trigger = { exists = title:e_hre }
			effect = {
				title:e_hre.holder = {
					if = {
						limit = {
							exists = character:73530
						}
						character:73530 = {
							save_scope_as = antipope_vassal
						}
					}
					else = {
						random_vassal_or_below = {
							limit = {
								tct_possible_antipope_vassal_trigger = yes
							}
							save_scope_as = antipope_vassal
						}
					}
					scope:antipope_vassal = {
						if = {
							limit = { 
								OR = {
									has_trait = cardinal_priest
									has_trait = cardinal_bishop
									has_trait = cardinal_archbishop
								}
							}
							if = { limit = { has_trait = cardinal_priest } remove_trait = cardinal_priest }
							if = { limit = { has_trait = cardinal_bishop } remove_trait = cardinal_bishop }
							if = { limit = { has_trait = cardinal_archbishop } remove_trait = cardinal_archbishop }
							remove_cardinal = yes
							promote_new_cardinal = yes
							update_cardinal_window = yes
							update_preferatus = yes
						}
						title:e_hre.holder = {
							culture = {
								save_scope_as = emperor_culture
							}
						}
						create_dynamic_title = {
							tier = duchy
							name = TCT_ANTIPAPCY
						}
						create_title_and_vassal_change = {
							type = created
							save_scope_as = change
							add_claim_on_loss = no
						}
						scope:new_title = {
							change_title_holder = {
								holder = scope:antipope_vassal
								change = scope:change
							}
							set_capital_county = title:e_hre.title_capital_county
							set_color_from_title = title:k_papal_state
							set_ruler_uses_title_name = no
							set_coa = title:k_papal_state
							set_definitive_form = yes
							set_can_be_named_after_dynasty = no
							set_landless_title = yes
							
							set_no_automatic_claims = yes
							set_always_follows_primary_heir = yes
							set_destroy_if_invalid_heir = yes
							copy_title_history = title:k_papal_state
						}
						change_liege = {
							liege = title:e_hre.holder
							change = scope:change
						}
				
						resolve_title_and_vassal_change = scope:change
					}
					title:e_hre.holder = {
						add_piety = massive_piety_loss
						struggle:investiture_controversy_struggle = {
							every_involved_ruler = {
								limit = {
									NOT = {
										has_character_flag = had_TCT_religious_events_0409
									}
								}
								trigger_event = { days = 1 id = TCT_religious_events.403 }
								add_character_flag = had_TCT_religious_events_0409
							}
						}
						faith.religious_head = {
							add_opinion = {
								modifier = TCT_creator_of_antipope
								target = title:e_hre.holder
							}
						}
					}
					TCT_create_antipope = {
						ANTIPOPE = scope:antipope_vassal
						CREATOR = title:e_hre.holder
					}
			
					title:e_hre.holder = {
						if = {
							limit = {
								title:e_hre.holder = {
									any_character_struggle	 = {
										involvement = involved
										phase_has_catalyst = tct_create_antipope
									}
								}
							}
							every_character_struggle = {
								involvement = involved
								activate_struggle_catalyst = {
									catalyst = tct_create_antipope
									character = title:e_hre.holder
								}
							}
						}
					}
				}
			}
		}

		triggered_effect = {
			trigger = { 
				NOT = { 
					exists = title:e_hre 
				} 
			}
			effect = {
				end_story = yes
			}
		}
	}
