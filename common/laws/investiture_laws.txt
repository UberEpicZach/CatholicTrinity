@investiture_cooldown_years = 10

investiture_law = { 
	default = papal_investiture
	cumulative = no
	flag = realm_law
	flag = investiture_law
	
	papal_investiture = {
		modifier = {}
		flag = papal_enacted
		can_keep = {
			faith = faith:catholic
			is_independent_ruler = yes
			highest_held_title_tier >= tier_duchy
		}

		can_pass = {
			faith = faith:catholic
			is_independent_ruler = yes
			NOR = {
				has_primary_title = title:k_papal_state
				has_variable = investiture_cooldown
			}
		}

		on_pass = {
			set_variable = {
				name = investiture_cooldown
				years = @investiture_cooldown_years
			}
		}

		ai_will_do = {
			value = investiture_law_ai_will_do_value
			if = {
				limit = { has_character_flag = TCT_start_investiture_controversy }
				subtract = 1000
			}
		}
	} 
	free_investiture = {
		modifier = {
			theocracy_government_tax_contribution_mult = 0.25
			religious_head_opinion = -50
		}
		flag = free_enacted
	
		can_keep = {
			faith = faith:catholic
			is_independent_ruler = yes
			highest_held_title_tier >= tier_duchy
		}

		can_pass = {
			faith = faith:catholic
			is_independent_ruler = yes
			trigger_if = {
				limit = { is_ai = yes }
				highest_held_title_tier >= tier_kingdom
			}
			trigger_else = {
				highest_held_title_tier >= tier_duchy
			}
		}
		
		pass_cost = {
			piety = {
				add = change_investiture_piety_cost
			}
			prestige = {
				add = change_investiture_prestige_cost
			}
		}
		
		on_pass = {
			set_variable = {
				name = investiture_cooldown
				years = @investiture_cooldown_years
			}
			if = {
				limit = {
					NOT = { has_realm_law = free_investiture }
				}
				hidden_effect = {	
					every_vassal = {
						limit = {
							OR = {
								any_vassal = {
									has_government = theocracy_government
								}
								has_government = theocracy_government
							}
						} 
						if = {
							limit = {
								any_vassal = {
									has_government = theocracy_government
								}
								has_government = theocracy_government
							}
							add_opinion = {
								target = root
								modifier = TCT_free_investiture
								opinion = 20
							}
						}
						else = {
							add_opinion = {
								target = root
								modifier = TCT_free_investiture
								opinion = 10
							}
						}
					}
				}
			}
		}
 
		ai_will_do = {
			value = investiture_law_ai_will_do_value
			multiply = -1
		}
	}
}
