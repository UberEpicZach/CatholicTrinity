tct_get_patriarchate_of_barony = {
	type = province
	text = {
		trigger = {
			NOT = {
				county.holder.top_liege.faith = {
					has_doctrine = tenet_communion
				}
				county.faith = {
					has_doctrine = tenet_communion
				}
			}
		}
		localization_key = tct_NO_PATRIARCHATE_TITLE
		#fallback = yes
	}
	text = {
		localization_key = tct_PATRIARCHATE_TITLE
		#fallback = yes
	}
}


tct_get_InvestitureLaw_of_barony = {
	type = province
	text = {
		trigger = {
			county.holder.top_liege = {
				has_realm_law = free_investiture
			}
		}
		localization_key = tct_FreeInvestitureMapMode
		#fallback = yes
	}
	text = {
		trigger = {
			county.holder.top_liege = {
				has_realm_law = papal_investiture
			}
		}
		localization_key = tct_PapalInvestitureMapMode
		#fallback = yes
	}
}

tct_get_patriarchate_of_pentarch = {
	type = character
	text = {
		trigger = {
			OR = {
				has_title = title:k_papal_state
				has_title = title:d_rome_patriarchate
			}
		}
		localization_key = tct_rom_title
	}
	text = {
		trigger = {
			has_title = title:k_orthodox
		}
		localization_key = tct_con_title
	}
	text = {
		trigger = {
			has_title = title:d_coptic_papacy
		}
		localization_key = tct_ale_title
	}
	text = {
		trigger = {
			has_title = title:d_antioch_patriarchate
		}
		localization_key = tct_ant_title
	}
	text = {
		trigger = {
			has_title = title:d_jerusalem_patriarchate
		}
		localization_key = tct_jer_title
	}
}
