#Setup values
holy_order_gold_cost = {
	value = 500
	if = {
		limit = {
			is_ai = yes
		}
		add = -300
	}
	# Remove half cost if there's a free credit available.
	if = {
		limit = { has_character_modifier = defender_of_the_faith }
		multiply = 0.1
	}
	# Remove cost if there's a free credit available.
	if = {
		limit = { has_character_modifier = next_free_ho_hire_modifier }
		multiply = 0
	}
	if = {
		limit = {
			has_trait = crusader_state_trait
		}
		add = -250
	}
}

holy_war_cb_piety_cost_multiplier = {
	value = 1

	if = {
		limit = {
			scope:attacker = {
				faith = { has_doctrine_parameter = cheaper_holy_wars_active }
			}
		}
		multiply = {
			value = 0.8
			desc = CB_REDUCED_HOLY_WAR_COST
		}
	}

	# Ghazi reduction
	if = {
		limit = {
			scope:attacker = {
				is_independent_ruler = no
				vassal_contract_has_flag = vassal_contract_cheaper_religious_war
			}
		}
		multiply = {
			value = ghazi_religious_war_cost_reduction
			desc = "CB_ATTACKER_VASSAL_CONTRACT"
		}
	}

	# Crusader state reduction
	if = {
		limit = {
			scope:attacker = {
				has_trait = crusader_state_trait
			}
		}
		multiply = {
			value = 0.9
			desc = CB_CRUSADER_STATE_COST
		}
	}
}
