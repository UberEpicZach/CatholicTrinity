}

saint_day_arrival_piety_gain = {
	value = 75
	if = { # Procession: Sizeable
		limit = {
			scope:activity = {
				has_activity_option = {
					category = tct_saint_days_procession_type
					option = tct_saint_days_procession_type_sizeable
				}
			}
		}
		add = 25
	}
	if = { # Procession: Grand
		limit = {
			scope:activity = {
				has_activity_option = {
					category = tct_saint_days_procession_type
					option = tct_saint_days_procession_type_grand
				}
			}
		}
		add = 75
	}
}

	value = 0
	if = {
		limit = {
			scope:activity = {

					option = tct_saint_days_cost_type_large
				}
			}
		}
		add = RICE_ancestor_max_dynasty_prestige_large_value
