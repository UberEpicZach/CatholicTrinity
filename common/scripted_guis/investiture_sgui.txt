can_see_investiture = {
    scope = character
	is_shown = {
		# has_realm_law = free_investiture
		has_faith = faith:catholic
	}
}

can_see_investiture_law = {
    scope = character
	is_shown = {
		OR = {
			has_realm_law = free_investiture_early
			has_realm_law = free_investiture_late
			top_liege = { has_realm_law = free_investiture_late }
			top_liege = { has_realm_law = free_investiture_early }
		}
	}
}

get_all_realm_theocracies = {
	scope = character
    effect = {
        primary_title = {
			save_scope_as = your_primary_title
			clear_variable_list = theocracy_vassals_for_investiture
		}
		ordered_vassal = {
			limit = { has_government = theocracy_government }
			order_by = highest_held_title_tier
			max = 9999
			check_range_bounds = no
			save_scope_as = to_add
			scope:your_primary_title = {
				add_to_variable_list = {
					name = theocracy_vassals_for_investiture
					target = scope:to_add
				}
			}
		}
    }
}

get_all_investiture_candidates = {
	effect = {
		save_scope_as = employer_liege_guy
		primary_title = {
			save_scope_as = title_thing
			set_variable = {
				name = selected_title_for_investiture
				value = scope:selected_title_inv
			}
			clear_variable_list = theocrat_candidates
		}
		set_variable = {
			name = selected_title_for_investiture
			value = scope:selected_title_inv
		}
		every_dynasty_member = {
			limit = {
				OR = {
					has_trait = devoted
					has_education_learning_trigger = yes
				}
				TCT_investiture_candidates_common_trigger = yes
			}
			save_scope_as = person_thing
			scope:title_thing = {
				add_to_variable_list = {
					name = theocrat_candidates
					target = scope:person_thing
				}
			}
		}
		every_sibling = {
			limit = {
				OR = {
					has_trait = devoted
					has_education_learning_trigger = yes
				}
				TCT_investiture_candidates_common_trigger = yes
			}
			save_scope_as = person_thing
			scope:title_thing = {
				add_to_variable_list = {
					name = theocrat_candidates
					target = scope:person_thing
				}
			}
		}
		every_child = {
			limit = {
				OR = {
					has_trait = devoted
					has_education_learning_trigger = yes
				}
				TCT_investiture_candidates_common_trigger = yes
			}
			save_scope_as = person_thing
			scope:title_thing = {
				add_to_variable_list = {
					name = theocrat_candidates
					target = scope:person_thing
				}
			}
		}
		
		every_vassal = {
			every_courtier_or_guest = {
				limit = {
					TCT_investiture_candidates_courtier_trigger = yes
				}
				save_scope_as = person_thing
				scope:title_thing = {
					add_to_variable_list = {
						name = theocrat_candidates
						target = scope:person_thing
					}
				}
			}
		}

		every_courtier_or_guest = {
			limit = {
				TCT_investiture_candidates_courtier_trigger = yes
			}
			save_scope_as = person_thing
			scope:title_thing = {
				add_to_variable_list = {
					name = theocrat_candidates
					target = scope:person_thing
				}
			}
		}
	}
}

set_up_new_successor = {
	is_shown = {
		has_variable = theocratic_successor
	}
	effect = {
		save_scope_as = theocratic_inheritor
		scope:title_inheritor = {
			set_variable = {
				name = theocratic_successor
				value = scope:theocratic_inheritor
			}
		}
		set_variable = {
			name = theocratic_successor
			value = scope:title_inheritor
		}
	}
}
